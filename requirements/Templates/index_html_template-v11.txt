    <script>
        async function connectAI() {
            const key = document.getElementById('apiKeyInput').value;
            const res = await fetch('/api/ai/connect', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({api_key: key})
            });
            const data = await res.json();
            if (data.status === 'success') {
                document.getElementById('aiStatusBadge').className = 'px-4 py-2 rounded-full text-xs font-bold bg-emerald-950 text-emerald-400 border border-emerald-500 neon-green';
                document.getElementById('aiStatusBadge').innerText = `AI: SYNCED (${data.latency_ms}ms)`;
            }
        }

        async function sendChatMessage() {
            const input = document.getElementById('chatInput');
            const msg = input.value;
            if (!msg) return;

            const chatLog = document.getElementById('chatHistoryLog');
            chatLog.innerHTML += `<div class="bg-blue-950/40 p-3 rounded-lg border border-blue-800/50 ml-auto max-w-[80%]"><span class="text-xs font-bold text-blue-400 block mb-1">Вы</span>${msg}</div>`;
            input.value = '';

            const res = await fetch('/api/ai/chat', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({message: msg})
            });
            const data = await res.json();
            chatLog.innerHTML += `<div class="bg-slate-800/50 p-3 rounded-lg border border-slate-700 max-w-[80%]"><span class="text-xs font-bold text-emerald-400 block mb-1">Antigravity Agent</span>${data.reply}</div>`;
            chatLog.scrollTop = chatLog.scrollHeight;
        }

        async function runAIBist() {
            const res = await fetch('/api/ai/run_bist', {method: 'POST'});
            const data = await res.json();
            document.getElementById('testAi1Status').innerText = data.results['TEST-AI-01'];
            document.getElementById('testAi1Status').className = 'text-emerald-400 font-bold';
            document.getElementById('testAi2Status').innerText = data.results['TEST-AI-02'];
            document.getElementById('testAi2Status').className = 'text-emerald-400 font-bold';
            document.getElementById('testAi3Status').innerText = data.results['TEST-AI-03'];
            document.getElementById('testAi3Status').className = 'text-emerald-400 font-bold';
        }
    </script>
