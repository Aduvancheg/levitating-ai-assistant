                Агент установлен на калибровочный стенд (Z = 30.0мм).
            </p>
            
            <div class="flex gap-4">
                <button id="btn-start" onclick="startCalib()" class="px-5 py-2.5 bg-cyan-600 hover:bg-cyan-500 font-semibold rounded-lg transition">
                    1. Запустить калибровку
                </button>
                <button id="btn-confirm" onclick="confirmRemoved()" disabled class="px-5 py-2.5 bg-emerald-600 hover:bg-emerald-500 disabled:opacity-40 font-semibold rounded-lg transition">
                    2. Подставка удалена (Confirm)

    <script>
        const ws = new WebSocket(`ws://${location.host}/ws/telemetry`);
        ws.onmessage = (event) => {
            const data = JSON.parse(event.data);
            if (data.calibration_fsm) {
                document.getElementById('fsm-badge').innerText = `FSM: ${data.calibration_fsm.state}`;
                document.getElementById('fsm-log').innerText = data.calibration_fsm.log;
                document.getElementById('btn-confirm').disabled = (data.calibration_fsm.state !== 'WAIT_REMOVAL');
            }
        };

        async function startCalib() {
            await fetch('/api/calibration/start', {method: 'POST'});
        }
        async function confirmRemoved() {
            await fetch('/api/calibration/confirm_stand_removed', {method: 'POST'});
        }
        async function cancelCalib() {
            await fetch('/api/calibration/cancel', {method: 'POST'});
        }
    </script>
