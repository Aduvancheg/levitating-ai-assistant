            })
            .then(res => res.json())
            .catch(err => console.error("Error retrying test:", err));
        }

        // --- DOUBLE CONFIRMATION BYPASS MODAL CONTROLS ---
        function openBypassModal() {
            document.getElementById('bypass-chk-agree').checked = false;
            document.getElementById('btn-bypass-confirm').disabled = true;
            document.getElementById('btn-bypass-confirm').className = "flex-1 py-2 bg-slate-800 text-slate-500 rounded text-xs font-bold cursor-not-allowed transition";
            document.getElementById('bypass-modal').classList.remove('hidden');
        }

        function closeBypassModal() {
            document.getElementById('bypass-modal').classList.add('hidden');
        }

        function toggleBypassSubmit() {
            const checked = document.getElementById('bypass-chk-agree').checked;
            const btn = document.getElementById('btn-bypass-confirm');
            if (checked) {
                btn.disabled = false;
                btn.className = "flex-1 py-2 bg-amber-600 hover:bg-amber-500 text-white rounded text-xs font-bold cursor-pointer transition glow-amber";
            } else {
                btn.disabled = true;
                btn.className = "flex-1 py-2 bg-slate-800 text-slate-500 rounded text-xs font-bold cursor-not-allowed transition";
            }
        }

        function confirmBypass() {
            fetch('/api/bist/bypass', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ override: true })
            })
            .then(res => res.json())
            .then(data => {
                closeBypassModal();
            })
            .catch(err => console.error("Bypass failed:", err));
        }

        function simulateFaultChange() {
            const selected = document.getElementById('fault-simulator-select').value;
            fetch('/api/bist/simulate_fault', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ test_id: selected === "NONE" ? null : selected })
            })
            .then(res => res.json())
            .catch(err => console.error("Simulator request failed:", err));
        }

        function confirmShutdown() {
            if (confirm("Вы действительно хотите безопасно выключить комплекс?\nСфера будет плавно опущена, катушки базы размагничены за 2 секунды для подавления импульсов самоиндукции, а Агент переведен в пассивный режим сна.")) {
                triggerPipeline('SAFE_SHUTDOWN');
            }
        }
